const GastoSchema = Schema({
    etiqueta: {
        type: String,
        required: true,
    },
    descripcion: {
        type: String,
        required: true,
    },
    valor: {
        type: Number,
        default: 0,
        required: true,
    },
    fecha: {
        type: Date,
        default: Date.now,
    },
    cuenta: {
        type: Schema.ObjectId,
        ref: "Cuenta"
    }
});

router.get("/prueba", ControladorGasto.pruebaGasto);
